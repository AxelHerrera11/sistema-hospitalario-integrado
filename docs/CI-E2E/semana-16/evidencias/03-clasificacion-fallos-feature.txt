# Script usado (copia literal):
    <?php
    // Clasifica los fallos de un JUnit de PHPUnit por módulo y causa (entorno sin PostgreSQL vs. aserción).
    $x = simplexml_load_file($argv[1]);
    $c = [];
    foreach ($x->xpath('//testcase[failure or error]') as $t) {
        $m = (string) ($t->error ?: $t->failure);
        $mod = explode(chr(92), (string) $t['class'])[2] ?? '?';
        $k = preg_match('/could not find driver|SQLSTATE\[08006\]|connection to server|pgsql|Connection refused/i', $m) ? 'requiere PostgreSQL'
            : (preg_match('/received 503/', $m) ? 'aserción: 503 (sin base de datos)' : 'aserción');
        $c[$mod][$k] = ($c[$mod][$k] ?? 0) + 1;
    }
    foreach ($c as $mod => $ks) {
        foreach ($ks as $k => $n) {
            printf("%-12s %3d  %s\n", $mod, $n, $k);
        }
    }
